import copy
import pytest
from subject import rotate_right
class IntChild(int): pass
class TupleChild(tuple): pass
def assert_exact_shape(actual, expected):
    assert type(actual) is type(expected)
    if type(expected) in (list,tuple):
        assert len(actual) == len(expected)
        for a,b in zip(actual,expected): assert_exact_shape(a,b)
    elif type(expected) is dict:
        assert actual.keys() == expected.keys()
        assert all(type(k) is str for k in actual)
        for key in expected: assert_exact_shape(actual[key],expected[key])
@pytest.mark.parametrize("args,expected", [
    pytest.param(((),0), ()),
    pytest.param(((1,2),0), (1,2)),
    pytest.param(((1,2),1), (2,1)),
    pytest.param(((1,2,3),1), (3,1,2)),
    pytest.param(((1,2,3),-1), (2,3,1)),
    pytest.param(((1,2,3),4), (1,2,3)),
    pytest.param(((1,2,3),-4), (1,2,3)),
    pytest.param((('a',),1), ('a',)),
    pytest.param((([1], [2]), 1), ([2], [1])),
    pytest.param((((), (1,2)), 1), ((1,2), ()),
    pytest.param(((None,),1), (None,)),
    pytest.param(((True,),1), (True,)),
    pytest.param(((IntChild(1),),1), (IntChild(1),)),
])
def test_values(args,expected):
    before=copy.deepcopy(args)
    actual=rotate_right(*args)
    assert actual == expected
    assert_exact_shape(actual,expected)
    assert args == before
@pytest.mark.parametrize("args", [
    pytest.param((TupleChild(),)),
    pytest.param(((1,),True)),
    pytest.param((TupleChild((1,2)),)),
    pytest.param(((),IntChild())),
])
def test_raises_typeerror(args):
    with pytest.raises(TypeError):
        rotate_right(*args)
